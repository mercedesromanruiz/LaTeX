\begin{itemize}

    \item Calidad

\subsection{Equipos}
\subsubsection{Principales}
\paragraph{Generadores}
Producen la potencia eléctrica.
\begin{itemize}
    \item Potencia nominal: $100\ -\ 700\ MW$
    \item Tensión nominal: $11\ -\ 21\ kV$
    \item Frecuencia/velocidad de giro:
    \begin{itemize}
        \item $50\ Hz / 3000\ rpm$
        \item $60\ Hz / 3600\ rpm$
    \end{itemize}
    \item Margen $\Delta \pm $tensión en bornas: $\pm 5\% \ -\ \pm 10\%$
    \item Excitación:
    \begin{itemize}
        \item Brushless (pequeños)
        \item Estática (grandes)
    \end{itemize}
    \item Refrigeración:
    \begin{itemize}
        \item TEWAC (pequeños)
        \item Hidrógeno (grandes)
    \end{itemize}
    \item Rango $\Delta \pm cos \phi$
    \item Precio: $F(Potencia)$
\end{itemize}
